\hypertarget{introduuxe7uxe3o}{%
\section{Introdu\c{c}\~{a}o}\label{introduuxe7uxe3o}}

\begin{frame}{Verifica\c{c}\~{a}o vs.~Infer\^{e}ncia}
\protect\hypertarget{verificauxe7uxe3o-vs.-inferuxeancia}{}
\textbf{Verifica\c{c}\~{a}o de tipos}: dado um programa \textbf{com anota\c{c}\~{o}es},
decide se é bem tipado.

\textbf{Infer\^{e}ncia de tipos}: dado um programa \textbf{sem anota\c{c}\~{o}es},
encontra os tipos mais gerais --- ou reporta que nenhum existe.

Regra (T-Abs) no STLC:

\[\frac{\Gamma,\, x : T_1 \vdash t : T_2}{\Gamma \vdash \lambda x {:} T_1.\; t : T_1 \to T_2}\]

\begin{itemize}
\tightlist
\item
  Verifica\c{c}\~{a}o: \(T_1\) é fornecido pelo programador
\item
  Infer\^{e}ncia: \(T_1\) deve ser \textbf{descoberto} a partir do uso de
  \(x\) no corpo
\end{itemize}
\end{frame}

\begin{frame}[fragile]{A Linguagem MiniML}
\protect\hypertarget{a-linguagem-miniml}{}
\begin{lstlisting}
e ::= x              -- variável
    | ℓ              -- literal (int ou bool)
    | e₁ e₂          -- aplica\c{c}\~{a}o
    | λx. e          -- abstra\c{c}\~{a}o sem anota\c{c}\~{a}o
    | let x = e₁ in e₂  -- defini\c{c}\~{a}o polimórfica
\end{lstlisting}

\begin{itemize}
\tightlist
\item
  Abstra\c{c}\~{o}es \textbf{n\~{a}o} t\^{e}m anota\c{c}\~{o}es de tipo
\item
  O inferidor deduz tudo automaticamente
\end{itemize}
\end{frame}

\begin{frame}{Tipos e Esquemas}
\protect\hypertarget{tipos-e-esquemas}{}
\textbf{Tipos monomórficos:}

\[T ::= \mathbf{Int} \mid \mathbf{Bool} \mid T \to T \mid \alpha\]

\textbf{Esquemas de tipo} (tipos polimórficos):

\[\sigma ::= T \mid \forall \vec{\alpha}.\; T\]

\begin{itemize}
\tightlist
\item
  Esquema \(\forall \alpha_1 \ldots \alpha_n.\; T\) generaliza \(T\) nas
  variáveis \(\alpha_i\)
\item
  A \textbf{instancia\c{c}\~{a}o} substitui variáveis quantificadas por tipos
  novos
\item
  A \textbf{generaliza\c{c}\~{a}o} de \(T\) em \(\Gamma\):
  \(\forall \vec{\alpha}.\; T\) onde
  \(\vec{\alpha} = \mathrm{fv}(T) \setminus \mathrm{fv}(\Gamma)\)
\end{itemize}
\end{frame}

\hypertarget{linguagem-de-restriuxe7uxf5es}{%
\section{Linguagem de Restri\c{c}\~{o}es}\label{linguagem-de-restriuxe7uxf5es}}

\begin{frame}{Gramática de Restri\c{c}\~{o}es}
\protect\hypertarget{gramuxe1tica-de-restriuxe7uxf5es}{}
\[\begin{array}{rcll}
C & ::= & \top & \text{trivial} \\
  & \mid & T_1 = T_2 & \text{equa\c{c}\~{a}o de tipos} \\
  & \mid & C_1 \wedge C_2 & \text{conjun\c{c}\~{a}o} \\
  & \mid & \exists \alpha.\; C & \text{variável existencial} \\
  & \mid & \mathbf{def}\; x : T\; \mathbf{in}\; C & \text{variável monomórfica} \\
  & \mid & \mathbf{let}\; x : T\; [C_1]\; \mathbf{in}\; C_2 & \text{defini\c{c}\~{a}o polimórfica} \\
  & \mid & \mathbf{inst}(x,\, T) & \text{instancia\c{c}\~{a}o}
\end{array}\]

\begin{itemize}
\tightlist
\item
  \(\exists \alpha.\; C\) --- variável de tipo local, visível apenas em
  \(C\)
\item
  \(\mathbf{def}\) --- par\^{a}metro de lambda (n\~{a}o polimórfico)
\item
  \(\mathbf{let}\) --- polimorfismo de let (generaliza após resolver
  \(C_1\))
\end{itemize}
\end{frame}

\begin{frame}{Gera\c{c}\~{a}o de Restri\c{c}\~{o}es \(\mathcal{G}(e, \alpha)\)}
\protect\hypertarget{gerauxe7uxe3o-de-restriuxe7uxf5es-mathcalge-alpha}{}
\(\mathcal{G}(e, \alpha)\) produz uma restri\c{c}\~{a}o cuja solu\c{c}\~{a}o torna
\(\alpha\) o tipo de \(e\):

\[\mathcal{G}(x,\; \alpha) = \mathbf{inst}(x,\, \alpha) \quad\text{(G-Var)}\]

\[\mathcal{G}(\ell,\; \alpha) = \alpha = \mathrm{tylit}(\ell) \quad\text{(G-Lit)}\]

\[\mathcal{G}(e_1\; e_2,\; \alpha) = \exists \beta.\;\bigl(\mathcal{G}(e_1,\; \beta \to \alpha) \wedge \mathcal{G}(e_2,\; \beta)\bigr) \quad\text{(G-App)}\]

\[\mathcal{G}(\lambda x.\; e,\; \alpha) = \exists \beta_1.\; \exists \beta_2.\; \bigl(\mathbf{def}\; x : \beta_1\; \mathbf{in}\; \mathcal{G}(e,\, \beta_2)\bigr) \wedge (\alpha = \beta_1 \to \beta_2) \quad\text{(G-Lam)}\]

\[\mathcal{G}(\mathbf{let}\; x = e_1\; \mathbf{in}\; e_2,\; \alpha) = \exists \beta.\; \mathbf{let}\; x : \beta\; [\mathcal{G}(e_1,\, \beta)]\; \mathbf{in}\; \mathcal{G}(e_2,\, \alpha) \quad\text{(G-Let)}\]
\end{frame}

\begin{frame}{Exemplo de Gera\c{c}\~{a}o}
\protect\hypertarget{exemplo-de-gerauxe7uxe3o}{}
Para \(\lambda f.\; f\; \mathbf{true}\) com alvo \(\alpha_0\):

\[\mathcal{G}(\lambda f.\; f\; \mathbf{true},\; \alpha_0) = \exists \beta_1\, \beta_2\, \beta_3.\; \bigl(\mathbf{def}\; f : \beta_1\; \mathbf{in}\; \mathbf{inst}(f,\, \beta_3 \to \beta_2) \wedge \beta_3 = \mathbf{Bool}\bigr) \wedge (\alpha_0 = \beta_1 \to \beta_2)\]

A resolu\c{c}\~{a}o for\c{c}a \(\beta_1 = \mathbf{Bool} \to \beta_2\).

Tipo principal: \(\alpha_0 = (\mathbf{Bool} \to \beta_2) \to \beta_2\),
generalizado em \(\beta_2\):

\[\forall \beta.\; (\mathbf{Bool} \to \beta) \to \beta\]
\end{frame}

\begin{frame}{Unifica\c{c}\~{a}o de Robinson}
\protect\hypertarget{unificauxe7uxe3o-de-robinson}{}
\(\mathrm{mgu}(T_1, T_2)\) encontra o \textbf{unificador mais geral}:

\[\begin{array}{rcll}
\mathrm{mgu}(\alpha, \alpha) & = & \varepsilon & \text{variável consigo mesma} \\
\mathrm{mgu}(\alpha, T) & = & [\alpha \mapsto T] & \text{se } \alpha \notin \mathrm{fv}(T) \\
\mathrm{mgu}(T_1 \to T_2,\; T_1' \to T_2') & = & S_2 \circ S_1 & S_1 = \mathrm{mgu}(T_1, T_1'),\; S_2 = \mathrm{mgu}(S_1(T_2), S_1(T_2')) \\
\mathrm{mgu}(T, T) & = & \varepsilon & \text{tipos concretos iguais} \\
\mathrm{mgu}(T_1, T_2) & = & \text{erro} & \text{caso contrário}
\end{array}\]

\textbf{Occurs check}: \(\alpha \notin \mathrm{fv}(T)\) --- evita tipos
infinitos como \(\alpha = \alpha \to \alpha\).
\end{frame}

\begin{frame}[fragile]{O Algoritmo de Resolu\c{c}\~{a}o}
\protect\hypertarget{o-algoritmo-de-resoluuxe7uxe3o}{}
\begin{lstlisting}[language=Haskell]
solve :: Constraint -> Solve ()
solve CTrue = pure ()
solve (t1 :=: t2) = do
    s  <- askSubst
    s' <- mgu (apply s t1) (apply s t2)
    extSubst s'
solve (c1 :&: c2) = solve c1 >> solve c2
solve (CExists c) = do
    v <- freshTyVar
    solve (c v)
solve (CDef x t c) =
    withLocalCtx x (Forall [] t) (solve c)
solve (CLet x t c1 c2) = do
    solve c1
    scheme <- generalize t
    withLocalCtx x scheme (solve c2)
solve (CInst x t) = do
    t' <- lookupVar x
    solve (t :=: t')
\end{lstlisting}
\end{frame}

\hypertarget{correuxe7uxe3o-e-completude}{%
\section{Corre\c{c}\~{a}o e Completude}\label{correuxe7uxe3o-e-completude}}

\begin{frame}{Teoremas Fundamentais}
\protect\hypertarget{teoremas-fundamentais}{}
\textbf{Teorema (Corre\c{c}\~{a}o)}: Se o resolvedor aceita \(C_e\) com
substitui\c{c}\~{a}o \(S\), ent\~{a}o \(\vdash e : S(\alpha_0)\).

\emph{O tipo produzido é válido segundo Hindley-Milner.}

\textbf{Teorema (Completude)}: Se \(\Gamma \vdash e : T\), ent\~{a}o o
resolvedor aceita \(C_e\) e produz um tipo do qual \(T\) é inst\^{a}ncia.

\emph{Se \(e\) é tipável, o algoritmo encontra o tipo mais geral.}

\textbf{Teorema (Tipo Principal)}: Existe \(\sigma\) tal que: 1.
\(\vdash e : T\) para todo \(T\) inst\^{a}ncia de \(\sigma\) 2. Todo tipo
válido \(T'\) é inst\^{a}ncia de \(\sigma\)

O algoritmo sempre encontra o \textbf{tipo principal}.
\end{frame}

Pipeline}\label{eliminauxe7uxe3o-de-tipos-e-pipeline}}

\begin{frame}{Elimina\c{c}\~{a}o de Tipos (Type Erasure)}
\protect\hypertarget{eliminauxe7uxe3o-de-tipos-type-erasure}{}
Tipos s\~{a}o necessários para análise estática, mas \textbf{desnecessários
em execu\c{c}\~{a}o}.

A fun\c{c}\~{a}o \(\lfloor \cdot \rfloor : \mathit{TyExp} \to \mathit{Term}\)
descarta anota\c{c}\~{o}es:

\[\begin{array}{rcl}
\lfloor x : T \rfloor & = & x \\
\lfloor (te_1\; te_2) : T \rfloor & = & \lfloor te_1 \rfloor\; \lfloor te_2 \rfloor \\
\lfloor \lambda x {:} T_1.\; te : T \rfloor & = & \lambda x.\; \lfloor te \rfloor \\
\lfloor \mathbf{let}\; x = te_1\; \mathbf{in}\; te_2 : T \rfloor & = & (\lambda x.\; \lfloor te_2 \rfloor)\; \lfloor te_1 \rfloor
\end{array}\]

\textbf{let} → beta-redex: \((\lambda x.\; t_2)\; t_1\) ---
semanticamente equivalente.
\end{frame}

\begin{frame}[fragile]{O Pipeline Completo de MiniML}
\protect\hypertarget{o-pipeline-completo-de-miniml}{}
\begin{lstlisting}[language=Haskell]
processTImp :: String -> IO ()
processTImp input =
  case parser input of
    Left err  -> putStrLn $ "Parse error: " ++ err
    Right ast ->
      case inferElab ast of
        Left err -> putStrLn $ "Type error: " ++ err
        Right (_, _, te, ty) -> do
          putStrLn $ "val : " ++ pretty ty    -- tipo exibido
          let lterm = erase te                -- elimina tipos
              prog  = lambdaToTImp lterm      -- closure conversion
          result <- interpret prog
          ...
\end{lstlisting}

\textbf{Quatro fases}: Parser → Infer\^{e}ncia/Elabora\c{c}\~{a}o → Elimina\c{c}\~{a}o →
TImp
\end{frame}

\begin{frame}[fragile]{Exemplo Completo}
\protect\hypertarget{exemplo-completo}{}
Para \passthrough{\lstinline!let f = \\x . x in f 42!}:

\begin{enumerate}
\item
  \textbf{Infer\^{e}ncia}: tipo \passthrough{\lstinline!Int!}, árvore tipada
  \(\mathbf{let}\; f = (\lambda x{:}\alpha.\; x){:}\alpha \to \alpha\; \mathbf{in}\; (f\; 42){:}\mathbf{Int}\)
\item
  \textbf{Elimina\c{c}\~{a}o}: \((\lambda f.\; f\; 42)\; (\lambda x.\; x)\)
\item
  \textbf{Closure conversion}: duas fun\c{c}\~{o}es promovidas,
  \passthrough{\lstinline!lam\_0!} (identidade) e
  \passthrough{\lstinline!lam\_1!} (aplica\c{c}\~{a}o de
  \passthrough{\lstinline!f!})
\item
  \textbf{Execu\c{c}\~{a}o TImp}: imprime \passthrough{\lstinline!42!}
\end{enumerate}
\end{frame}

\begin{frame}{Sumário do Capítulo}
\protect\hypertarget{sumuxe1rio-do-capuxedtulo}{}
\begin{itemize}
\tightlist
\item
  \textbf{Infer\^{e}ncia de tipos} resolve um problema mais difícil que

\item
  \textbf{Gera\c{c}\~{a}o de restri\c{c}\~{o}es} \(\mathcal{G}(e, \alpha)\) transforma

\item
  \textbf{Polimorfismo de let}: generaliza\c{c}\~{a}o após resolver restri\c{c}\~{o}es
  do corpo
\item
  \textbf{Elimina\c{c}\~{a}o de tipos}: descarta anota\c{c}\~{o}es, reutiliza pipeline
  do \(\lambda\)-cálculo
\end{itemize}
\end{frame}
