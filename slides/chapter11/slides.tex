\section{Introdução ao Happy}\label{introduuxe7uxe3o-ao-happy}

\begin{frame}{Motivação}

    \begin{itemize}

        \item
        Construir tabelas LALR à mão é impraticável para gramáticas reais
        \item
        Assim como o \textbf{Alex} automatiza análise léxica, o \textbf{Happy}
        automatiza análise sintática
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Motivação}

    \begin{itemize}

        \item
        Happy gera analisadores \textbf{LALR(1)} a partir de especificações
        gramaticais no estilo BNF
        \item
        O próprio compilador \textbf{GHC} usa um parser gerado pelo Happy
        \item
        Análogo ao \texttt{yacc} e
        \texttt{Bison} do ecossistema C/C++
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Estrutura de um Arquivo
    \texttt{.y}}

    \begin{lstlisting}
        { cabeçalho Haskell }

        diretivas e declaração de tokens

        %%

        regras gramaticais

        { rodapé Haskell }
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Estrutura de um Arquivo
    \texttt{.y}}

    \begin{itemize}

        \item
        Quatro partes separadas pelo marcador \texttt{\%\%}:
    \end{itemize}

    \begin{enumerate}

        \item
        \textbf{Cabeçalho}: módulo e imports Haskell (copiado literalmente)
        \item
        \textbf{Diretivas}: configurações do parser e declaração de tokens
        \item
        \textbf{Regras}: produções BNF com ações semânticas em Haskell
        \item
        \textbf{Rodapé}: funções auxiliares
    \end{enumerate}
\end{frame}

\begin{frame}[fragile]{Gerando o Parser}

    \begin{lstlisting}[language=bash]
        happy ExpParser.y            # gera ExpParser.hs
        happy -i ExpParser.y         # também gera ExpParser.info (tabela)
        happy -o ExpParser.hs ExpParser.y
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Gerando o Parser}

    \begin{itemize}

        \item
        Com Cabal: basta listar \texttt{happy} em
        \texttt{build-tool-depends}
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Diretivas}

    \begin{lstlisting}
        %name parser Exp
        %monad {Alex}{(>>=)}{return}
        %tokentype { Token }
        %error     { parseError }
        %lexer {lexer}{Token _ TEOF}
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Diretivas}

    \begin{itemize}

        \item
        \texttt{\%monad} integra com a mônada
        \texttt{Alex} do lexer gerado pelo Alex
        \item
        \texttt{\%lexer} faz Happy chamar
        \texttt{lexer} para obter cada token (modo
        incremental)
        \item
        O modo incremental é necessário quando o lexer é monádico
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Declaração de Tokens}

    \begin{itemize}

        \item
        A seção \texttt{\%token} associa nomes simbólicos a
        padrões de correspondência:
    \end{itemize}

    \begin{lstlisting}
        %token
        num       {Token _ (TNumber $$)}
        '('       {Token _ TLParen}
        ')'       {Token _ TRParen}
        '+'       {Token _ TPlus}
        '*'       {Token _ TTimes}
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Declaração de Tokens}

    \begin{itemize}

        \item
        \texttt{$$} em
        \texttt{Token \_ (TNumber $$)} captura o valor
        inteiro
    \end{itemize}

    --- Disponível nas ações como \texttt{$1},
    \texttt{$2}, etc.

    \begin{itemize}

        \item
        Tokens sem valor (como \texttt{TLParen}) apenas
        verificam o construtor
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Declarações de Precedência}

    \begin{lstlisting}
        %left '+'
        %left '*'
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Declarações de Precedência}

    \begin{itemize}

        \item
        A \textbf{ordem} das linhas determina precedência: de menor para maior
        \item
        \texttt{*} aparece depois de
        \texttt{+} -> \texttt{*} tem
        \textbf{maior} precedência
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Declarações de Precedência}

    \begin{itemize}

        \item
        \texttt{\%left}: associatividade à esquerda
        \item
        \texttt{\%right}: associatividade à direita
        \item
        \texttt{\%nonassoc}: não associativo (erro se
        encadeado)
    \end{itemize}
\end{frame}


\section{Regras Gramaticais e
Ações}\label{regras-gramaticais-e-auxe7uxf5es}

\begin{frame}[fragile]{Regras Gramaticais}

    \begin{itemize}

        \item
        Apos o \texttt{\%\%}, produções BNF com ações em
        Haskell:
    \end{itemize}

    \begin{lstlisting}
        Exp : num         { EInt $1 }
        | Exp '+' Exp { $1 :+: $3 }
        | Exp '*' Exp { $1 :*: $3 }
        | '(' Exp ')' { $2 }
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Regras Gramaticais}

    \begin{itemize}

        \item
        \texttt{$1}, \texttt{$2},
        \texttt{$3}: valores semânticos dos símbolos (da
        esquerda para a direita)
        \item
        Ação executada quando o parser \textbf{reduz} por aquela produção
        \item
        Tipo inferido pelo compilador a partir das ações:
        \texttt{Exp}
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Rodapé: Funções Auxiliares}

    \begin{lstlisting}[language=Haskell]
        {
        lexer :: (Token -> Alex a) -> Alex a
        lexer = (=<< alexMonadScan)

        expParser :: String -> IO (Either String Exp)
        expParser content = pure $ runAlex content parser
        }
    \end{lstlisting}
\end{frame}

\section{Conflitos}\label{conflitos}

\begin{frame}[fragile]{Shift-Reduce}

    \begin{itemize}

        \item
        Ocorre quando o parser pode tanto fazer shift quanto reduce no mesmo
        estado e token.
    \end{itemize}

    \textbf{Causa comum}: gramática de expressões escrita de forma ambígua:

    \begin{lstlisting}
        Exp : num
        | Exp '+' Exp
        | Exp '*' Exp
        | '(' Exp ')'
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Shift-Reduce}

    \begin{itemize}

        \item
        Para \texttt{1 + 2 * 3}, após empilhar
        \texttt{Exp '+' Exp} (com
        \texttt{Exp} = \texttt{2}) e ver
        \texttt{*}:

        \begin{itemize}

            \item
            \textbf{Reduce} \texttt{Exp '+' Exp} -> interpreta
            \texttt{(1+2)*3}
            \item
            \textbf{Shift} \texttt{*} -> interpreta
            \texttt{1+(2*3)} (correto!)
        \end{itemize}
        \item
        Happy retorna:
        \texttt{ExpParser.y: shift/reduce conflicts: 4}
    \end{itemize}
\end{frame}

\begin{frame}{Solução: Precedência}

    Happy usa as declarações para resolver:

    \begin{itemize}

        \item
        Token com \textbf{maior} precedência que a produção -> \textbf{shift}
        \item
        Token com \textbf{menor} precedência -> \textbf{reduce}
        \item
        Token com \textbf{mesma} precedência -> aplica associatividade
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Shift-Reduce}

    \begin{itemize}

        \item
        Gramática para if/else
    \end{itemize}

    \begin{lstlisting}
        Stmt : 'if' Exp 'then' Stmt
        | 'if' Exp 'then' Stmt 'else' Stmt
        | 'other'
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Shift-Reduce}

    \begin{itemize}

        \item
        Para \texttt{if e1 then if e2 then s1 else s2}:

        \begin{itemize}

            \item
            Shift \texttt{else} -> pertence ao
            \texttt{if} mais próximo (interno): convenção
            padrão
            \item
            Reduce \texttt{if ... then s1} ->
            \texttt{else} pertence ao
            \texttt{if} externo
        \end{itemize}
        \item
        O comportamento padrão do Happy (preferir shift) resolve esse problema
        corretamente.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Reduce-Reduce}

    \begin{itemize}

        \item
        Ocorre quando dois itens completos diferentes reivindicam a mesma
        entrada.
    \end{itemize}

    \begin{lstlisting}
        Prog : Decl
        | Expr

        Decl : 'id'
        Expr : 'id'
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Reduce-Reduce}

    Ao ver \texttt{id} e precisar reduzir:

    \begin{itemize}

        \item
        Não sabe se deve criar \texttt{Decl} ou
        \texttt{Expr}
        \item
        Happy escolhe a \textbf{primeira regra} (aviso, não erro)
    \end{itemize}
\end{frame}

\begin{frame}{Reduce-Reduce}

    \begin{itemize}

        \item
        Diferentemente do shift-reduce, conflitos reduce-reduce raramente têm
        resolução padrão correta.
    \end{itemize}

    --- \textbf{Quase sempre indicam uma gramática problemática}.
\end{frame}

\begin{frame}[fragile]{Reduce-Reduce}

    \begin{itemize}

        \item
        Solução: Tornar os dois não-terminais sintáticos distintos por
        contexto:
    \end{itemize}

    \begin{lstlisting}
        -- Antes: ambíguo
        Prog : 'id'   -- pode ser Decl ou Expr?

        -- Depois: discriminar por palavra-chave
        Prog : 'let' 'id'    -- declaração
        | 'id'          -- expressão
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Reduce-Reduce}

    \begin{itemize}

        \item
        Outra abordagem: remover ambiguidade na análise semântica
    \end{itemize}

    --- Exemplo: \texttt{typedef} de C, onde
    identificadores de tipo requerem consulta ao contexto.
\end{frame}
