\begin{frame}{Como executamos programas?}
\protect\hypertarget{como-executamos-programas}{}
\begin{itemize}

\item
  Tr\^{e}s técnicas principais

  \begin{itemize}

  \item
    Interpreta\c{c}\~{a}o
  \item
    Compila\~{c}\~{a}o
  \item
    Virtualiza\c{c}\~{a}o.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Interpreta\c{c}\~{a}o}
\protect\hypertarget{interpretauxe7uxe3o}{}
\begin{itemize}

\item
  Código fonte é representado como uma estrutura de dados e o
  intepretador executa o programa diretamente.
\item
  Interpretador é responsável pela execu\c{c}\~{a}o do programa e de sua
  intera\c{c}\~{a}o com o S.O.
\end{itemize}
\end{frame}

\begin{frame}{Compila\~{c}\~{a}o}
\protect\hypertarget{compilauxe7uxe3o}{}
\begin{itemize}

\item
  Código fonte é traduzido para uma nova linguagem.
\item
  Código alvo pode ser uma linguagem de alto nível (transpila\c{c}\~{a}o).
\item
  Código alvo pode ser uma linguagem de baixo nível (compila\~{c}\~{a}o).
\end{itemize}
\end{frame}

\begin{frame}{Virtualiza\c{c}\~{a}o}
\protect\hypertarget{virtualizauxe7uxe3o}{}
\begin{itemize}

\item
  Código fonte é traduzido para uma linguagem de baixo nível, n\~{a}o
  diretamente entendida pelo hardware.
\item
  Normalmente, chamado de bytecode.
\item
  Intepretadores mais eficientes.
\end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}
\protect\hypertarget{escalando-a-montanha}{}
\begin{itemize}

\item
  Análise Léxica: Dividir o código em \textbf{tokens}.

  \begin{itemize}

  \item
    Eliminar comentários e espa\c{c}os em branco.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}
\protect\hypertarget{escalando-a-montanha-1}{}
\begin{itemize}

\item
  Análise sintática: Organiza a sequ\^{e}ncia de tokens em sua estrutura
  gramatical.

  \begin{itemize}

  \item
    Produz uma árvore de sintaxe abstrata.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}
\protect\hypertarget{escalando-a-montanha-2}{}
\begin{itemize}

\item
  Análise sem\^{a}ntica: Verifica se a árvore produzida pelo analisador
  sintático atende restri\c{c}\~{o}es sem\^{a}nticas.

  \begin{itemize}

  \item
    Todos os símbolos foram declarados?
  \item
    Argumentos de fun\c{c}\~{o}es possuem a aridade e tipo corretos?
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}
\protect\hypertarget{escalando-a-montanha-3}{}
\begin{itemize}

\item
  Intepretador: executa o código diretamente a partir da árvore de
  sintaxe.

  \begin{itemize}

  \item
    N\~{a}o há gera\c{c}\~{a}o de código
  \item
    Comum em linguagens dinamicamente tipadas, como Python.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}
\protect\hypertarget{escalando-a-montanha-4}{}
\begin{itemize}

\item
  ``Transpilador'': converte a árvore de sintaxe para outra linguagem de
  alto nível.

  \begin{itemize}

  \item
    Ideal para permitir independ\^{e}ncia de plataformas.
  \item
    Exemplo: Typescript.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}
\protect\hypertarget{escalando-a-montanha-5}{}
\begin{itemize}

\item
  Gera\c{c}\~{a}o de código intermediário: código de baixo nível independente de
  plataforma.

  \begin{itemize}

  \item
    Utilizado para otimiza\c{c}\~{o}es independentes de hardware.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}
\protect\hypertarget{escalando-a-montanha-6}{}
\begin{itemize}

\item
  Otimiza\c{c}\~{a}o de código.

  \begin{itemize}

  \item
    Otimiza\c{c}\~{a}o de consumo de memória, tempo de execu\c{c}\~{a}o, tamanho de
    código, etc\ldots{}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}
\protect\hypertarget{escalando-a-montanha-7}{}
\begin{itemize}

\item
  Gera\c{c}\~{a}o de código: tradu\c{c}\~{a}o da representa\c{c}\~{a}o intermediária para o
  código de máquina, diretamente executável pelo hardware.
\end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}
\protect\hypertarget{escalando-a-montanha-8}{}
\begin{itemize}

\item
  Virtualiza\c{c}\~{a}o: gera\c{c}\~{a}o de código para máquinas virtuais como a JVM /
  EVM.

  \begin{itemize}

  \item
    Diferentes plataformas podem executar o mesmo programa utilizando
    VMs para a plataforma.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Front-end de um compilador}
\protect\hypertarget{front-end-de-um-compilador}{}
\begin{itemize}

\item
  Responsável pela análise do código.
\item
  Produz uma representa\c{c}\~{a}o intermediária para gera\c{c}\~{a}o de código.
\end{itemize}
\end{frame}

\begin{frame}{Middle-end de um compilador}
\protect\hypertarget{middle-end-de-um-compilador}{}
\begin{itemize}

\item
  Responsável por otimiza\c{c}\~{o}es independentes de arquitetura.

  \begin{itemize}

  \item
    Constant folding: substituir 3 + 4 por 7 no código.
  \item
    Dead code elimination: eliminar código que nunca é executado.
  \item
    Loop unrolling: eliminar la\c{c}os para reduzir custo de desvios.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Back-end de um compilador}
\protect\hypertarget{back-end-de-um-compilador}{}
\begin{itemize}

\item
  Responsável por otimiza\c{c}\~{o}es dependentes do hardware.

  \begin{itemize}

  \item
    Aloca\c{c}\~{a}o de registradores: como usar registradores da melhor
    maneira?
  \item
    Escalonamento de instru\c{c}\~{o}es: qual a ordem de instru\c{c}\~{o}es para melhor
    efici\^{e}ncia?
  \item
    Otimiza\c{c}\~{o}es específicas de arquitetura: usar vectoriza\c{c}\~{a}o, múltiplos
    núcleos.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Porque essa arquitetura?}
\protect\hypertarget{porque-essa-arquitetura}{}
\begin{itemize}

\item
  Um mesmo front-end pode suportar diferentes middle-ends, sem
  modifica\c{c}\~{o}es na linguagem fonte.
\item
  Um mesmo middle-end pode suportar diferentes arquiteturas.
\item
  Separa\c{c}\~{a}o permite que desenvolvimento seja focado em problemas de cada
  componente.
\end{itemize}
\end{frame}
