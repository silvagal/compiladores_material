\section{Compiladores}\label{compiladores}}

\begin{frame}{Como executamos programas?}

    \begin{itemize}

        \item
        Três técnicas principais

        \begin{itemize}

            \item
            Interpretação
            \item
            Compilação
            \item
            Virtualização.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Interpretação}

    \begin{itemize}

        \item
        Código fonte é representado como uma estrutura de dados e o
        intepretador executa o programa diretamente.
        \item
        Interpretador é responsável pela execução do programa e de sua
        interação com o S.O.
    \end{itemize}
\end{frame}

\begin{frame}{Compilação}

    \begin{itemize}

        \item
        Código fonte é traduzido para uma nova linguagem.
        \item
        Código alvo pode ser uma linguagem de alto nível (transpilação).
        \item
        Código alvo pode ser uma linguagem de baixo nível (compilação).
    \end{itemize}
\end{frame}

\begin{frame}{Virtualização}

    \begin{itemize}

        \item
        Código fonte é traduzido para uma linguagem de baixo nível, não
        diretamente entendida pelo hardware.
        \item
        Normalmente, chamado de bytecode.
        \item
        Intepretadores mais eficientes.
    \end{itemize}
\end{frame}

\begin{frame}{Visão geral (montanha)}

    \includegraphics[width=\textwidth,height=0.82\textheight,keepaspectratio]{apresentacao/montain.png}
\end{frame}

\begin{frame}{Escalando a montanha}

    \begin{itemize}

        \item
        Análise Léxica: Dividir o código em \textbf{tokens}.

        \begin{itemize}

            \item
            Eliminar comentários e espaços em branco.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}

    \begin{itemize}

        \item
        Análise sintática: Organiza a sequência de tokens em sua estrutura
        gramatical.

        \begin{itemize}

            \item
            Produz uma árvore de sintaxe abstrata.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}

    \begin{itemize}

        \item
        Análise semântica: Verifica se a árvore produzida pelo analisador
        sintático atende restrições semânticas.

        \begin{itemize}

            \item
            Todos os símbolos foram declarados?
            \item
            Argumentos de funções possuem a aridade e tipo corretos?
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}

    \begin{itemize}

        \item
        Intepretador: executa o código diretamente a partir da árvore de
        sintaxe.

        \begin{itemize}

            \item
            Não há geração de código
            \item
            Comum em linguagens dinamicamente tipadas, como Python.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}

    \begin{itemize}

        \item
        ``Transpilador'': converte a árvore de sintaxe para outra linguagem de
        alto nível.

        \begin{itemize}

            \item
            Ideal para permitir independência de plataformas.
            \item
            Exemplo: Typescript.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}

    \begin{itemize}

        \item
        Geração de código intermediário: código de baixo nível independente de
        plataforma.

        \begin{itemize}

            \item
            Utilizado para otimizações independentes de hardware.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}

    \begin{itemize}

        \item
        Otimização de código.

        \begin{itemize}

            \item
            Otimização de consumo de memória, tempo de execução, tamanho de
            código, etc\ldots{}
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}

    \begin{itemize}

        \item
        Geração de código: tradução da representação intermediária para o
        código de máquina, diretamente executável pelo hardware.
    \end{itemize}
\end{frame}

\begin{frame}{Escalando a montanha}

    \begin{itemize}

        \item
        Virtualização: geração de código para máquinas virtuais como a JVM /
        EVM.

        \begin{itemize}

            \item
            Diferentes plataformas podem executar o mesmo programa utilizando
            VMs para a plataforma.
        \end{itemize}
    \end{itemize}
\end{frame}

\section{Estrutura de um compilador}\label{estrutura-de-um-compilador}}

\begin{frame}{Estrutura geral}

    \begin{itemize}

        \item
        Estrutura geral de um compilador
    \end{itemize}

    \includegraphics[width=\textwidth,height=0.82\textheight,keepaspectratio]{apresentacao/compiler_structure.png}
\end{frame}

\begin{frame}{Front-end de um compilador}

    \begin{itemize}

        \item
        Responsável pela análise do código.
        \item
        Produz uma representação intermediária para geração de código.
    \end{itemize}
\end{frame}

\begin{frame}{Middle-end de um compilador}

    \begin{itemize}

        \item
        Responsável por otimizações independentes de arquitetura.

        \begin{itemize}

            \item
            Constant folding: substituir 3 + 4 por 7 no código.
            \item
            Dead code elimination: eliminar código que nunca é executado.
            \item
            Loop unrolling: eliminar laços para reduzir custo de desvios.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Back-end de um compilador}

    \begin{itemize}

        \item
        Responsável por otimizações dependentes do hardware.

        \begin{itemize}

            \item
            Alocação de registradores: como usar registradores da melhor
            maneira?
            \item
            Escalonamento de instruções: qual a ordem de instruções para melhor
            eficiência?
            \item
            Otimizações específicas de arquitetura: usar vectorização, múltiplos
            núcleos.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Porque essa arquitetura?}

    \begin{itemize}

        \item
        Um mesmo front-end pode suportar diferentes middle-ends, sem
        modificações na linguagem fonte.
        \item
        Um mesmo middle-end pode suportar diferentes arquiteturas.
        \item
        Separação permite que desenvolvimento seja focado em problemas de cada
        componente.
    \end{itemize}
\end{frame}
